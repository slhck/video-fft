 99    def radial_profile(data: np.ndarray, center: tuple[int, int]) -> np.ndarray:
100        """
101        Calculate the radial profile of a 2D array.
102
103        Args:
104            data (np.ndarray): 2D array
105            center (tuple[int, int]): center of the radial profile
106
107        Returns:
108            np.ndarray: radial profile
109
110        Source: https://stackoverflow.com/a/21242776/435093
111        """
112        y, x = np.indices((data.shape))
113        r = np.sqrt((x - center[0]) ** 2 + (y - center[1]) ** 2)
114        r = r.astype(int)
115
116        tbin = np.bincount(r.ravel(), data.ravel())
117        nr = np.bincount(r.ravel())
118        radialprofile = tbin / nr
119        return radialprofile

222    def calc_fft(self) -> None:
223        """
224        Calculate FFT-based metrics for the input video.
225
226        Processes each frame by computing its 2D FFT, then extracts the azimuthally
227        averaged 1D power spectrum (radial profile) from the magnitude spectrum.
228        The sum of high-frequency components is used as a metric to detect upscaled
229        content - native resolution videos have higher values than upscaled ones.
230
231        Results are stored in self.data as a VideoFftData dict containing:
232        - input_file: absolute path to input
233        - average_fft: mean of high-frequency sums across all frames
234        - max_fft, min_fft, median_fft: distribution statistics
235        - pct_05, pct_95: 5th and 95th percentile values
236
237        If first_frame, all_frames, or mean options are set, corresponding
238        magnitude spectrum images are saved to the output directory.
239
240        Raises:
241            RuntimeError: If no video streams found or no FFT values calculated.
242        """
243        # Generate empty list fft sum values
244        fft_values: list[int] = []
245
246        if self.num_frames is not None:
247            num_frames = min(self.get_num_frames(), self.num_frames)
248        else:
249            num_frames = self.get_num_frames()
250
251        t = tqdm(total=num_frames, disable=self.quiet, file=sys.stderr)
252
253        current_frame = 0
254        for frame_data in self._read_input_file():
255            fshift = np.fft.fftshift(np.fft.fft2(frame_data))
256            self.last_magnitude_spectrum = 20 * np.log(np.abs(fshift))
257
258            if self.avg_magnitude_spectrum is None:
259                self.avg_magnitude_spectrum = self.last_magnitude_spectrum
260            else:
261                self.avg_magnitude_spectrum = np.mean(
262                    np.array(
263                        [self.avg_magnitude_spectrum, self.last_magnitude_spectrum]
264                    ),
265                    axis=0,
266                )
267
268            if not (self.frame_height or self.frame_width):
269                self.frame_height, self.frame_width = self.last_magnitude_spectrum.shape
270
271            if self.frame_height is None or self.frame_width is None:
272                raise RuntimeError("Could not determine frame width/height")
273
274            if (self.first_frame and current_frame == 0) or self.all_frames:
275                self._plot_magnitude_spectrum("last", current_frame)
276
277            center = (int(self.frame_width / 2), int(self.frame_height / 2))
278
279            # Calculate the azimuthally averaged 1D power spectrum
280            psf_1d = VideoFftCalculator.radial_profile(
281                cast(np.ndarray, self.last_magnitude_spectrum), center
282            )
283            # Get the sum of the high frequencies
284            low_freq_ind = int((len(psf_1d) / 2))
285            fft_values.append(sum(psf_1d[low_freq_ind:]))
286
287            current_frame += 1
288            t.update()
289            if num_frames is not None and current_frame >= num_frames:
290                break
291
292        t.close()
293
294        if self.mean:
295            self._plot_magnitude_spectrum("mean")
296
297        if not fft_values:
298            raise RuntimeError("No FFT values were calculated!")
299
300        self.data = {
301            "input_file": os.path.abspath(self.input_file),
302            "average_fft": int(np.mean(fft_values)),
303            "max_fft": int(max(fft_values)),
304            "min_fft": int(min(fft_values)),
305            "median_fft": int(np.median(fft_values)),
306            "pct_05": int(np.percentile(fft_values, 5, method="midpoint")),
307            "pct_95": int(np.percentile(fft_values, 95, method="midpoint")),
308        }
